Rencanakan sebelum menulis kode

1 jam

Tulis dulu apa yang mau dibuat, supaya tidak bongkar-pasang di tengah jalan.

  • Fitur: daftar/login, katalog + cari, keranjang, checkout, pembayaran, admin.
  • Aturan penting: stok tidak boleh minus, tidak boleh bayar dua kali, harga pesanan tidak berubah walau harga produk berubah.
  • Data: users, products, categories, carts, orders, payments (lihat ERD di dokumen rekayasa).
  • Endpoint: tulis daftar URL seperti POST /auth/login, GET /products, POST /orders.
  • Git flow: branch main (rilis), develop (kerja), feature/nama-fitur per fitur.
Cek berhasil: Daftar fitur, tabel, dan endpoint di catatan.

// validate fails fast on configuration that would make the app insecure or
// unable to start, instead of discovering it on the first request.
func (c *Config) validate() error {
    if len(c.JWT.Secret) < 32 {
        return errors.New("JWT_SECRET must be set and at least 32 characters long")
    }
    if c.JWT.AccessTTL <= 0 || c.JWT.RefreshTTL <= 0 {
        return errors.New("JWT_ACCESS_TTL and JWT_REFRESH_TTL must be positive durations")
    }
    if c.App.BcryptCost < 4 || c.App.BcryptCost > 31 {
        return errors.New("BCRYPT_COST must be between 4 and 31")
    }
    if c.IsProduction() && strings.TrimSpace(c.CORS.AllowOrigins) == "*" {
        return errors.New("CORS_ALLOW_ORIGINS must not be '*' in production")
    }
    return nil
}

// Migrate applies every pending SQL migration embedded in the binary.
//
// It uses its own short-lived connection because golang-migrate closes the
// connection it is given when it finishes.
func Migrate(dsn string) error {
    db, err := sql.Open("pgx", dsn)
    if err != nil {
        return fmt.Errorf("open migration connection: %w", err)
    }
    defer db.Close()

    driver, err := migratepgx.WithInstance(db, &migratepgx.Config{})
    if err != nil {
        return fmt.Errorf("create migration driver: %w", err)
    }

    source, err := iofs.New(migrations.FS, ".")
    if err != nil {
        return fmt.Errorf("load migration files: %w", err)
    }

    m, err := migrate.NewWithInstance("iofs", source, "postgres", driver)
    if err != nil {
        return fmt.Errorf("init migrate: %w", err)
    }

    if err := m.Up(); err != nil && !errors.Is(err, migrate.ErrNoChange) {
        return fmt.Errorf("apply migrations: %w", err)
    }
    return nil
}

// Connect opens the GORM connection and configures the connection pool.
func Connect(cfg config.DBConfig, log *zap.Logger) (*gorm.DB, error) {
    db, err := gorm.Open(postgres.Open(cfg.DSN()), &gorm.Config{
        // GORM's own logger is silenced; errors surface through the
        // application error handler with a request ID attached.
        Logger: gormlogger.Default.LogMode(gormlogger.Silent),
        // TranslateError maps driver errors to gorm.ErrDuplicatedKey etc.,
        // so repositories do not depend on PostgreSQL error codes.
        TranslateError: true,
        NowFunc:        func() time.Time { return time.Now().UTC() },
    })
    if err != nil {
        return nil, fmt.Errorf("open database: %w", err)
    }

    sqlDB, err := db.DB()
    if err != nil {
        return nil, fmt.Errorf("get sql.DB: %w", err)
    }
    sqlDB.SetMaxOpenConns(cfg.MaxOpenConns)
    sqlDB.SetMaxIdleConns(cfg.MaxIdleConns)
    sqlDB.SetConnMaxLifetime(30 * time.Minute)

    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    if err := sqlDB.PingContext(ctx); err != nil {
        return nil, fmt.Errorf("ping database: %w", err)
    }

    log.Info("database connected", zap.String("host", cfg.Host), zap.String("name", cfg.Name))
    return db, nil
}

type txKey struct{}

// TxManager lets the service layer define a transaction boundary without
// importing GORM. The transaction travels inside the context, and every
// repository picks it up through FromContext.
type TxManager struct {
    db *gorm.DB
}

func NewTxManager(db *gorm.DB) *TxManager { return &TxManager{db: db} }

// WithinTx runs fn inside one database transaction. Returning an error (or
// panicking) rolls everything back; returning nil commits.
func (m *TxManager) WithinTx(ctx context.Context, fn func(ctx context.Context) error) error {
    return m.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
        return fn(context.WithValue(ctx, txKey{}, tx))
    })
}

// FromContext returns the transaction stored in ctx, or fallback when the
// call is not part of a transaction.
func FromContext(ctx context.Context, fallback *gorm.DB) *gorm.DB {
    if tx, ok := ctx.Value(txKey{}).(*gorm.DB); ok {
        return tx
    }
    return fallback.WithContext(ctx)
}

// Auth verifies the Bearer access token and stores the caller's identity in
// the request context. Missing or invalid tokens produce 401.
func Auth(tokens *token.Manager) fiber.Handler {
    return func(c *fiber.Ctx) error {
        header := c.Get(fiber.HeaderAuthorization)
        scheme, value, found := strings.Cut(header, " ")
        if !found || !strings.EqualFold(scheme, "Bearer") || strings.TrimSpace(value) == "" {
            return apperror.Unauthorized("Missing or malformed Authorization header")
        }

        claims, err := tokens.Parse(strings.TrimSpace(value))
        if err != nil {
            return apperror.Unauthorized("Invalid or expired token")
        }
        userID, err := claims.UserID()
        if err != nil {
            return apperror.Unauthorized("Invalid or expired token")
        }

        c.Locals(localUserID, userID)
        c.Locals(localRole, claims.Role)
        return c.Next()
    }
}

// RateLimit is a fixed-window limiter backed by Redis, keyed by client IP.
//
// It fails open: when Redis is unavailable the request is allowed, because a
// cache outage must not take the whole API down.
func RateLimit(store cache.Cache, max int, window time.Duration, log *zap.Logger) fiber.Handler {
    return func(c *fiber.Ctx) error {
        key := "ratelimit:" + c.IP()
        count, err := store.IncrWithTTL(c.UserContext(), key, window)
        if err != nil {
            log.Warn("rate limiter unavailable, allowing request", zap.Error(err))
            return c.Next()
        }

        remaining := max - int(count)
        if remaining < 0 {
            remaining = 0
        }
        c.Set("X-RateLimit-Limit", strconv.Itoa(max))
        c.Set("X-RateLimit-Remaining", strconv.Itoa(remaining))

        if int(count) > max {
            c.Set(fiber.HeaderRetryAfter, strconv.Itoa(int(window.Seconds())))
            return apperror.TooManyRequests("Too many requests, please try again later")
        }
        return c.Next()
    }
}

// Refresh rotates the refresh token: the presented token is revoked and a
// new pair is issued.
//
// If a token that was already revoked is presented again, it was either
// replayed by an attacker or by the legitimate user after theft. Both cases
// are handled the same way: every session of that user is revoked.
func (s *Service) Refresh(ctx context.Context, req RefreshRequest) (*TokenResponse, error) {
    invalid := apperror.Unauthorized("Invalid or expired refresh token")

    stored, err := s.tokens.FindByHash(ctx, hashToken(req.RefreshToken))
    if errors.Is(err, apperror.ErrNotFound) {
        return nil, invalid
    }
    if err != nil {
        return nil, apperror.Internal(err)
    }
    if time.Now().After(stored.ExpiresAt) {
        return nil, invalid
    }

    revoked, err := s.tokens.Revoke(ctx, stored.ID)
    if err != nil {
        return nil, apperror.Internal(err)
    }
    if !revoked {
        if err := s.tokens.RevokeAllForUser(ctx, stored.UserID); err != nil {
            return nil, apperror.Internal(err)
        }
        return nil, invalid
    }

    u, err := s.users.FindByID(ctx, stored.UserID)
    if errors.Is(err, apperror.ErrNotFound) {
        return nil, invalid
    }
    if err != nil {
        return nil, apperror.Internal(err)
    }
    return s.issueTokens(ctx, u)
}

// Get returns a product using the cache-aside strategy:
//
//  1. Try Redis.
//  2. On a miss (or any Redis error) read PostgreSQL.
//  3. Store the result in Redis with a TTL for the next reader.
//
// Redis errors are logged and otherwise ignored: the cache may only ever make
// a request faster, never make it fail.
func (s *Service) Get(ctx context.Context, id uuid.UUID) (*Response, error) {
    key := detailKey(id)

    var cached Response
    found, err := s.cache.Get(ctx, key, &cached)
    if err != nil {
        s.log.Warn("product cache read failed", zap.Error(err))
    }
    if found {
        return &cached, nil
    }

    p, err := s.repo.FindByID(ctx, id)
    if errors.Is(err, apperror.ErrNotFound) {
        return nil, apperror.NotFound("Product not found")
    }
    if err != nil {
        return nil, apperror.Internal(err)
    }

    res := ToResponse(p)
    if err := s.cache.Set(ctx, key, res, s.detailTTL); err != nil {
        s.log.Warn("product cache write failed", zap.Error(err))
    }
    return &res, nil
}

// DecreaseStock atomically reserves qty units and returns the updated row.
//
// The whole check-and-decrement is a single statement:
//
//    UPDATE products SET stock = stock - $qty
//    WHERE id = $id AND stock >= $qty AND deleted_at IS NULL
//    RETURNING *
//
// PostgreSQL takes a row lock for the UPDATE, so two concurrent checkouts
// cannot both pass the "stock >= qty" condition for the last unit. When no
// row is affected the product is out of stock (or gone).
func (r *GormRepository) DecreaseStock(ctx context.Context, id uuid.UUID, qty int) (*Product, error) {
    var p Product
    res := database.FromContext(ctx, r.db).Model(&p).
        Clauses(clause.Returning{}).
        Where("id = ? AND stock >= ?", id, qty).
        Updates(map[string]any{
            "stock":      gorm.Expr("stock - ?", qty),
            "updated_at": time.Now().UTC(),
        })
    if res.Error != nil {
        return nil, res.Error
    }
    if res.RowsAffected == 0 {
        return nil, ErrInsufficientStock
    }
    return &p, nil
}

// ToResponse computes subtotals and the cart total from current prices.
// Totals are derived on every read instead of being stored, so they can
// never drift from the items.
func ToResponse(c *Cart) Response {
    res := Response{ID: c.ID, Items: make([]ItemResponse, 0, len(c.Items))}
    for _, it := range c.Items {
        exists := it.Product.ID != uuid.Nil
        line := ItemResponse{
            ID:          it.ID,
            ProductID:   it.ProductID,
            ProductName: it.Product.Name,
            ProductSlug: it.Product.Slug,
            UnitPrice:   it.Product.Price,
            Quantity:    it.Quantity,
            Subtotal:    it.Product.Price * int64(it.Quantity),
            Available:   exists && it.Product.Stock >= it.Quantity,
        }
        res.Items = append(res.Items, line)
        res.TotalItems += it.Quantity
        res.TotalAmount += line.Subtotal
    }
    return res
}

// AddItem adds a product to the cart. Adding a product that is already in
// the cart increases its quantity. The resulting quantity may not exceed the
// available stock.
//
// This check is a courtesy to the user, not the oversell protection: stock
// can still change before checkout, where it is enforced atomically.
func (s *Service) AddItem(ctx context.Context, userID uuid.UUID, req AddItemRequest) (*Cart, error) {
    productID, err := uuid.Parse(req.ProductID)
    if err != nil {
        return nil, apperror.BadRequest("Invalid product_id")
    }

    p, err := s.products.FindByID(ctx, productID)
    if errors.Is(err, apperror.ErrNotFound) {
        return nil, apperror.NotFound("Product not found")
    }
    if err != nil {
        return nil, apperror.Internal(err)
    }
    if p.Stock <= 0 {
        return nil, apperror.Conflict("Product is out of stock")
    }

    c, err := s.Get(ctx, userID)
    if err != nil {
        return nil, err
    }

    quantity := req.Quantity
    for _, it := range c.Items {
        if it.ProductID == productID {
            quantity += it.Quantity
            break
        }
    }
    if quantity > p.Stock {
        return nil, apperror.Conflict(fmt.Sprintf("Only %d unit(s) of this product are available", p.Stock))
    }

    if err := s.repo.UpsertItem(ctx, c.ID, productID, quantity); err != nil {
        return nil, apperror.Internal(err)
    }
    return s.Get(ctx, userID)
}

// Create converts the user's cart into an order.
//
// Everything runs in one transaction: reserving stock for every item,
// inserting the order with its item snapshots, and clearing the cart. If any
// step fails (for example one product is out of stock) the whole checkout is
// rolled back and no stock is lost.
//
// Prices and names come from the row returned by the stock update itself,
// never from the request, so the client cannot influence what it pays.
func (s *Service) Create(ctx context.Context, userID uuid.UUID) (*Order, error) {
    var created *Order
    var productIDs []uuid.UUID

    err := s.tx.WithinTx(ctx, func(ctx context.Context) error {
        c, err := s.carts.GetForCheckout(ctx, userID)
        if err != nil {
            return err
        }
        if len(c.Items) == 0 {
            return apperror.BadRequest("Cart is empty")
        }

        // Always lock product rows in the same order. Two checkouts that
        // share products would otherwise be able to deadlock each other.
        items := append([]cart.Item(nil), c.Items...)
        sort.Slice(items, func(i, j int) bool {
            return items[i].ProductID.String() < items[j].ProductID.String()
        })

        order := &Order{
            UserID: userID,
            Status: StatusPending,
            Items:  make([]Item, 0, len(items)),
        }
        for _, it := range items {
            p, err := s.stock.DecreaseStock(ctx, it.ProductID, it.Quantity)
            if errors.Is(err, product.ErrInsufficientStock) {
                name := it.Product.Name
                if name == "" {
                    name = "A product in your cart"
                }
                return apperror.Conflict(fmt.Sprintf("%s is out of stock or no longer available in the requested quantity", name))
            }
            if err != nil {
                return err
            }

            productID := p.ID
            subtotal := p.Price * int64(it.Quantity)
            order.Items = append(order.Items, Item{
                ProductID:   &productID,
                ProductName: p.Name,
                UnitPrice:   p.Price,
                Quantity:    it.Quantity,
                Subtotal:    subtotal,
            })
            order.TotalAmount += subtotal
            productIDs = append(productIDs, p.ID)

// Create starts a payment for a PENDING order.
//
// The Idempotency-Key header makes the call safe to retry: sending the same
// key again returns the payment created the first time (created = false)
// instead of creating a second one. The unique index on idempotency_key is
// what guarantees this under concurrency; the lookup before the insert is
// only the fast path.
func (s *Service) Create(ctx context.Context, caller Caller, idempotencyKey string, req CreateRequest) (p *Payment, created bool, err error) {
    key := strings.TrimSpace(idempotencyKey)
    if key == "" {
        return nil, false, apperror.BadRequest("Idempotency-Key header is required")
    }
    if len(key) > 100 {
        return nil, false, apperror.BadRequest("Idempotency-Key must be at most 100 characters")
    }
    orderID, err := uuid.Parse(req.OrderID)
    if err != nil {
        return nil, false, apperror.BadRequest("Invalid order_id")
    }

    o, err := s.ownedOrder(ctx, caller, orderID)
    if err != nil {
        return nil, false, err
    }

    if existing, err := s.replay(ctx, key, orderID); err != nil || existing != nil {
        return existing, false, err
    }

    if o.Status != order.StatusPending {
        return nil, false, apperror.Conflict(fmt.Sprintf("Order with status %s cannot be paid", o.Status))
    }
    if existing, err := s.activePayment(ctx, orderID, key); err != nil || existing != nil {
        return existing, false, err
    }

    ref, err := newReference()
    if err != nil {
        return nil, false, apperror.Internal(err)
    }
    p = &Payment{
        OrderID:        orderID,
        Reference:      ref,
        IdempotencyKey: key,
        Amount:         o.TotalAmount,
        Status:         StatusPending,
        ExpiresAt:      time.Now().UTC().Add(s.expiry),
    }

    err = s.repo.Create(ctx, p)
    if errors.Is(err, apperror.ErrDuplicate) {
        // Lost a race. Either the same key was inserted concurrently (replay
        // it) or the order already has another active payment.
        if existing, rerr := s.replay(ctx, key, orderID); rerr != nil || existing != nil {
            return existing, false, rerr
        }
        return nil, false, apperror.Conflict("Order already has an active payment")
    }
    if err != nil {
        return nil, false, apperror.Internal(err)
    }
    return p, true, nil
}

/// Stores tokens in the platform keystore (Android Keystore, iOS Keychain).
/// SharedPreferences would keep them in plain text.
///
/// The access token is also cached in memory because it is read on every
/// request and keystore reads are comparatively slow.
class SecureTokenStorage implements TokenStorage {
  SecureTokenStorage(this._storage);

  final FlutterSecureStorage _storage;
  String? _accessToken;
  bool _loaded = false;

  static const _accessKey = 'access_token';
  static const _refreshKey = 'refresh_token';

  @override
  Future<String?> readAccessToken() async {
    if (!_loaded) {
      _accessToken = await _storage.read(key: _accessKey);
      _loaded = true;
    }
    return _accessToken;
  }

  @override
  Future<String?> readRefreshToken() => _storage.read(key: _refreshKey);

  @override
  Future<void> save({required String accessToken, required String refreshToken}) async {
    _accessToken = accessToken;
    _loaded = true;
    await _storage.write(key: _accessKey, value: accessToken);
    await _storage.write(key: _refreshKey, value: refreshToken);
  }

  @override
  Future<void> clear() async {
    _accessToken = null;
    _loaded = true;
    await _storage.delete(key: _accessKey);
    await _storage.delete(key: _refreshKey);
  }
}

# CGO is disabled to get a static binary; -s -w strips debug information.

FROM alpine:3.20

    && addgroup -S app && adduser -S app -G app
